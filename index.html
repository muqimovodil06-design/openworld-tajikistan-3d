// УПРАВЛЕНИЕ WASD + ANDROID const keys = {}; const keyMap = { KeyA: 'left', KeyD: 'right', KeyW: 'forward', KeyS: 'back' }; window.addEventListener('keydown', e => { if (keyMap[e.code]) { keys[keyMap[e.code]] = true; e.preventDefault(); } }); window.addEventListener('keyup', e => { if (keyMap[e.code]) { keys[keyMap[e.code]] = false; e.preventDefault(); } }); ['left','right','forward','back'].forEach(id => { const btn = document.getElementById(id); btn.addEventListener('pointerdown', e => { e.preventDefault(); keys[id] = true; }); ['pointerup','pointerleave','pointercancel'].forEach(event => { btn.addEventListener(event, () => keys[id] = false); }); });
